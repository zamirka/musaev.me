{{- /* Опыт работы на общей оси времени: строка на организацию. Строки независимы, поэтому пересекающиеся периоды (в ранние годы работа, преподавание и учёба шли одновременно) видны сами собой, без упаковки по дорожкам. */ -}} {{- $now := now -}} {{- $nowM := add (mul $now.Year 12) (int ($now.Format "1")) -}} {{- $segs := slice -}} {{- range hugo.Data.cv.positions -}} {{- $f := time.AsTime (printf "%s-01T00:00:00Z" (string .from)) -}} {{- $s := add (mul $f.Year 12) (int ($f.Format "1")) -}} {{- $e := $nowM -}} {{- $ty := 0 -}} {{- if .to -}} {{- $t := time.AsTime (printf "%s-01T00:00:00Z" (string .to)) -}} {{- $e = add (mul $t.Year 12) (int ($t.Format "1")) -}} {{- $ty = $t.Year -}} {{- end -}} {{- $segs = $segs | append (dict "label" (.short | default .org) "org" .org "role" .role "s" $s "e" $e "now" (not .to) "y0" $f.Year "y1" $ty) -}} {{- end -}} {{- $segs = sort $segs "s" -}} {{- /* Периоды в одной организации сводим в одну строку. */ -}} {{- $rows := slice -}} {{- range $seg := $segs -}} {{- $found := false -}} {{- $next := slice -}} {{- range $r := $rows -}} {{- if eq $r.org $seg.org -}} {{- $found = true -}} {{- $next = $next | append (merge $r (dict "s" (cond (lt $seg.s $r.s) $seg.s $r.s) "e" (cond (gt $seg.e $r.e) $seg.e $r.e) "y0" (cond (lt $seg.y0 $r.y0) $seg.y0 $r.y0) "y1" (cond (gt $seg.y1 $r.y1) $seg.y1 $r.y1) "now" (or $r.now $seg.now))) -}} {{- else -}} {{- $next = $next | append $r -}} {{- end -}} {{- end -}} {{- $rows = cond $found $next ($rows | append $seg) -}} {{- end -}} {{- $min := (index $segs 0).s -}} {{- $span := sub $nowM $min -}} {{- $startYear := (index $segs 0).y0 -}} {{- $rows = sort $rows "e" "desc" -}}
